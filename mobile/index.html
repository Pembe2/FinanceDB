function computeRemainingAtDeath(portfolioAtRetirement, growthRate, yearsToDeath) { const r = growthRate / 100; // Convert percentage growth rate to decimal const n = yearsToDeath; // Years after retirement until death to calculate // Calculate constant annual withdrawal const withdrawal = portfolioAtRetirement * r / (1 - Math.pow(1 + r, -n)); return withdrawal; // Output constant withdrawal amount }
